// ── 解析 hyp 字符串 → [{speaker, start, end, text}] ──
function parseHyp(hyp) {
  if (!hyp || hyp === 'none') return [];
//   const re = /\[(\d{2}:\d{2}\.\d{2})\s*-->\s*(\d{2}:\d{2}\.\d{2})\]\s*(Speaker \d+):\s*(.*)/;
  const re = new RegExp("\\[(\\d{2}:\\d{2}\\.\\d{2})\\s*-->" +
                      "\\s*(\\d{2}:\\d{2}\\.\\d{2})\\]\\s*(Speaker \\d+):\\s*(.*)");
  return hyp.trim().split('\n').map(line => {
    const m = line.trim().match(re);
    if (!m) return null;
    return {
      start: parseTS(m[1]),
      end:   parseTS(m[2]),
      speaker: m[3].trim(),
      text: m[4].trim()
    };
  }).filter(Boolean).sort((a,b) => a.start - b.start);
}

function parseTS(s) {
  const [mm, ss] = s.split(':');
  return parseFloat(mm) * 60 + parseFloat(ss);
}

// ── 检测重叠 ──
function detectOverlaps(lines) {
  const flags = new Array(lines.length).fill(false);
  for (let i = 0; i < lines.length; i++) {
    for (let j = i + 1; j < lines.length; j++) {
      if (lines[i].start < lines[j].end - 0.1 && lines[i].end > lines[j].start + 0.1) {
        flags[i] = flags[j] = true;
      }
    }
  }
  return flags;
}

function fmtTime(s) {
  const m = Math.floor(s / 60), sec = Math.floor(s % 60);
  return m + ':' + String(sec).padStart(2, '0');
}

// ════════════════════════════════════════════════════════════════
// 🔧 核心：实时高亮同步（rAF 驱动，支持多行重叠同时高亮）
// ════════════════════════════════════════════════════════════════

demoInstances.forEach(inst => {
  const media = document.getElementById(inst.mediaId);
  const scrollEl = document.getElementById(inst.scrollId);
  if (!media || !scrollEl) return;

  let prevKey = '';      // 上一帧的 active 集合签名
  let userScrolling = false;
  let scrollTimer = null;

  // 检测用户手动滚动 → 暂停自动滚动 2 秒
  scrollEl.addEventListener('scroll', () => {
    userScrolling = true;
    clearTimeout(scrollTimer);
    scrollTimer = setTimeout(() => { userScrolling = false; }, 2000);
  }, { passive: true });

  let rafId = null;

  function sync() {
    const t = media.currentTime;
    const subLines = scrollEl.querySelectorAll('.sub-line');
    if (!subLines.length) { rafId = requestAnimationFrame(sync); return; }

    // ── 找所有当前激活的行（支持重叠同时高亮） ──
    const activeSet = new Set();
    for (let i = 0; i < inst.lines.length; i++) {
      if (t >= inst.lines[i].start && t < inst.lines[i].end) {
        activeSet.add(i);
      }
    }
    // 静默区间：fallback 到最近已说过的行
    if (activeSet.size === 0) {
      for (let i = inst.lines.length - 1; i >= 0; i--) {
        if (t >= inst.lines[i].start) { activeSet.add(i); break; }
      }
    }

    // 签名比较，跳过无变化帧
    const key = [...activeSet].join(',');
    if (key !== prevKey) {
      prevKey = key;
      subLines.forEach((el, i) => {
        el.classList.toggle('active', activeSet.has(i));
      });

      // 自动滚动到最早的激活行
      if (!userScrolling && activeSet.size > 0) {
        const earliest = Math.min(...activeSet);
        if (subLines[earliest]) {
          subLines[earliest].scrollIntoView({ behavior: 'smooth', block: 'nearest' });
        }
      }
    }

    if (!media.paused) {
      rafId = requestAnimationFrame(sync);
    }
  }

  media.addEventListener('play', () => { if (!rafId) rafId = requestAnimationFrame(sync); });
  media.addEventListener('pause', () => { cancelAnimationFrame(rafId); rafId = null; sync(); });
  media.addEventListener('seeked', () => { prevKey = ''; sync(); });

  // 初始状态
  sync();
});

(function () {
  const canvas = document.getElementById('bg-wave');
  const ctx = canvas.getContext('2d');
  function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
  window.addEventListener('resize', resize); resize();

  const PAL = ['rgba(124,92,252,','rgba(0,212,255,','rgba(255,107,157,','rgba(74,222,128,','rgba(245,200,66,'];
  const rng = (seed => () => { seed = (seed * 16807 + 0) % 2147483647; return (seed - 1) / 2147483646; })(42);

  const WAVES = Array.from({ length: 10 }, (_, i) => ({
    yPct: 0.04 + i * 0.093,
    freq: 0.007 + rng() * 0.013,
    amp: 14 + rng() * 32,
    freq2: 0.018 + rng() * 0.020,
    amp2f: 0.20 + rng() * 0.20,
    speed: 0.12 + rng() * 0.32,
    phase: rng() * Math.PI * 2,
    color: PAL[i % PAL.length],
    alpha: 0.055 + rng() * 0.085,
    lw: 0.7 + rng() * 1.1,
  }));

  const PARTICLES = Array.from({ length: 45 }, () => {
    const c = PAL[Math.floor(rng() * PAL.length)];
    const bvx = (rng() - 0.5) * 0.28, bvy = (rng() - 0.5) * 0.28;
    return {
      x: rng() * window.innerWidth, y: rng() * window.innerHeight,
      r: 0.8 + rng() * 2.2, vx: bvx, vy: bvy, baseVx: bvx, baseVy: bvy,
      alpha: 0.18 + rng() * 0.35, color: c, pulse: rng() * Math.PI * 2,
    };
  });

  const mouse = { x: -9999, y: -9999, moving: false };
  let _mt = null;
  window.addEventListener('mousemove', e => {
    mouse.x = e.clientX; mouse.y = e.clientY; mouse.moving = true;
    clearTimeout(_mt); _mt = setTimeout(() => { mouse.moving = false; }, 150);
  });

  let t = 0;
  function draw() {
    const W = canvas.width, H = canvas.height;
    ctx.clearRect(0, 0, W, H);

    WAVES.forEach(w => {
      const cy = H * w.yPct;
      ctx.beginPath();
      ctx.strokeStyle = w.color + w.alpha + ')';
      ctx.lineWidth = w.lw;
      ctx.lineJoin = 'round';
      for (let x = 0; x <= W; x += 4) {
        const y = cy
          + Math.sin(x * w.freq + t * w.speed + w.phase) * w.amp
          + Math.sin(x * w.freq2 + t * w.speed * 0.55) * (w.amp * w.amp2f);
        x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
      }
      ctx.stroke();
    });

    PARTICLES.forEach(p => {
      const a = p.alpha * (0.7 + 0.3 * Math.sin(t * 0.8 + p.pulse));
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
      ctx.fillStyle = p.color + a + ')';
      ctx.fill();
      const mdx = mouse.x - p.x, mdy = mouse.y - p.y;
      const md = Math.sqrt(mdx * mdx + mdy * mdy);
      if (md < 160 && md > 0) {
        const f = (1 - md / 160) * 0.065;
        const nx = mdx / md, ny = mdy / md;
        if (mouse.moving) { p.vx -= nx * f; p.vy -= ny * f; }
        else {
          const OR = 35;
          if (md > OR) { p.vx += nx * f; p.vy += ny * f; }
          else { p.vx -= nx * f * 1.5; p.vy -= ny * f * 1.5; }
          p.vx += -ny * f * 0.45; p.vy += nx * f * 0.45;
        }
      }
      p.vx = p.vx * 0.92 + p.baseVx * 0.08;
      p.vy = p.vy * 0.92 + p.baseVy * 0.08;
      const spd = Math.sqrt(p.vx * p.vx + p.vy * p.vy);
      if (spd > 3.5) { p.vx = p.vx / spd * 3.5; p.vy = p.vy / spd * 3.5; }
      p.x += p.vx; p.y += p.vy;
      if (p.x < -10) p.x = W + 10; if (p.x > W + 10) p.x = -10;
      if (p.y < -10) p.y = H + 10; if (p.y > H + 10) p.y = -10;
    });

    t += 0.028;
    requestAnimationFrame(draw);
  }
  draw();
})();
